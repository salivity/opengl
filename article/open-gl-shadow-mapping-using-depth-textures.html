OpenGL Shadow Mapping Using Depth Textures

Shadow mapping in OpenGL is a two-pass rendering technique that simulates dynamic shadows by capturing scene depth from the perspective of a light source and comparing it during the camera render pass. In the first pass, the scene is rendered from the light's point of view into a custom framebuffer object (FBO) backed by a depth texture. In the second pass, the scene is rendered normally from the camera's perspective, where a fragment shader projects coordinates into light space to sample the depth texture and determine whether a fragment is in shadow.

1. Framebuffer and Depth Texture Setup

To store the depth values computed from the light source, you must create a dedicated Framebuffer Object (FBO) and attach a 2D depth texture.

GLuint depthMapFBO;
glGenFramebuffers(1, &depthMapFBO);

const GLuint SHADOW_WIDTH = 1024, SHADOW_HEIGHT = 1024;
GLuint depthMap;
glGenTextures(1, &depthMap);
glBindTexture(GL_TEXTURE_2D, depthMap);
glTexImage2D(GL_TEXTURE_2D, 0, GL_DEPTH_COMPONENT, 
            SHADOW_WIDTH, SHADOW_HEIGHT, 0, GL_DEPTH_COMPONENT, GL_FLOAT, NULL);

glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_BORDER);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_BORDER);
float borderColor[] = { 1.0f, 1.0f, 1.0f, 1.0f };
glTexParameterfv(GL_TEXTURE_2D, GL_TEXTURE_BORDER_COLOR, borderColor);

glBindFramebuffer(GL_FRAMEBUFFER, depthMapFBO);
glFramebufferTexture2D(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT, GL_TEXTURE_2D, depthMap, 0);
glDrawBuffer(GL_NONE);
glReadBuffer(GL_NONE);
glBindFramebuffer(GL_FRAMEBUFFER, 0);

Using GL_CLAMP_TO_BORDER with a border color of 1.0 ensures that any coordinates sampled outside the light projection frustum have maximum depth, preventing false shadows outside the light's range.

2. Pass 1: Rendering the Depth Map

Before rendering, compute the light space transformation matrix by multiplying a projection matrix by a view matrix set to the light's position and orientation. For directional lights, an orthographic projection is used; for spotlights, use a perspective projection.

glm::mat4 lightProjection = glm::ortho(-10.0f, 10.0f, -10.0f, 10.0f, 1.0f, 25.0f);
glm::mat4 lightView = glm::lookAt(lightPos, glm::vec3(0.0f), glm::vec3(0.0f, 1.0f, 0.0f));
glm::mat4 lightSpaceMatrix = lightProjection * lightView;

glViewport(0, 0, SHADOW_WIDTH, SHADOW_HEIGHT);
glBindFramebuffer(GL_FRAMEBUFFER, depthMapFBO);
glClear(GL_DEPTH_BUFFER_BIT);

// Render scene using the depth shader
depthShader.use();
depthShader.setMat4("lightSpaceMatrix", lightSpaceMatrix);
RenderScene(depthShader);

glBindFramebuffer(GL_FRAMEBUFFER, 0);

Depth Shader Code

The vertex shader transforms vertices into light space:

// depth_vertex.glsl
#version 330 core
layout (location = 0) in vec3 aPos;

uniform mat4 lightSpaceMatrix;
uniform mat4 model;

void main()
{
    gl_Position = lightSpaceMatrix * model * vec4(aPos, 1.0);
}

The fragment shader can remain completely empty because OpenGL automatically updates the depth buffer based on the vertex clip coordinates:

// depth_fragment.glsl
#version 330 core
void main() {}

3. Pass 2: Rendering the Scene with Shadows

Reset the viewport to your window dimensions, bind the generated depth map texture to a texture unit, and render the scene using your primary lighting shaders.

glViewport(0, 0, WINDOW_WIDTH, WINDOW_HEIGHT);
glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

mainShader.use();
mainShader.setMat4("projection", cameraProjection);
mainShader.setMat4("view", cameraView);
mainShader.setMat4("lightSpaceMatrix", lightSpaceMatrix);

glActiveTexture(GL_TEXTURE1);
glBindTexture(GL_TEXTURE_2D, depthMap);
mainShader.setInt("shadowMap", 1);

RenderScene(mainShader);

4. Shadow Calculation in the Main Shader

The main fragment shader projects the current fragment into light space, reads the stored depth from the depth texture, and tests if the current fragment's depth is greater than the recorded depth.

Vertex Shader

Pass the fragment's position in light space to the fragment shader:

#version 330 core
layout (location = 0) in vec3 aPos;
layout (location = 1) in vec3 aNormal;

out vec3 FragPos;
out vec3 Normal;
out vec4 FragPosLightSpace;

uniform mat4 model;
uniform mat4 view;
uniform mat4 projection;
uniform mat4 lightSpaceMatrix;

void main()
{
    FragPos = vec3(model * vec4(aPos, 1.0));
    Normal = mat3(transpose(inverse(model))) * aNormal;
    FragPosLightSpace = lightSpaceMatrix * vec4(FragPos, 1.0);
    gl_Position = projection * view * vec4(FragPos, 1.0);
}

Fragment Shader

Normalize the coordinates, apply a depth bias to mitigate shadow acne, and optionally apply Percentage-Closer Filtering (PCF) for smoother edges:

#version 330 core
out vec4 FragColor;

in vec3 FragPos;
in vec3 Normal;
in vec4 FragPosLightSpace;

uniform sampler2D shadowMap;
uniform vec3 lightPos;
uniform vec3 viewPos;

float CalculateShadow(vec4 fragPosLightSpace, vec3 normal, vec3 lightDir)
{
    // Perform perspective divide
    vec3 projCoords = fragPosLightSpace.xyz / fragPosLightSpace.w;
    // Transform from [-1, 1] range to [0, 1] texture coordinates
    projCoords = projCoords * 0.5 + 0.5;

    if (projCoords.z > 1.0)
        return 0.0;

    // Calculate bias based on surface angle to eliminate shadow acne
    float bias = max(0.05 * (1.0 - dot(normal, lightDir)), 0.005);

    // Percentage-Closer Filtering (PCF) for soft edges
    float shadow = 0.0;
    vec2 texelSize = 1.0 / textureSize(shadowMap, 0);
    for(int x = -1; x <= 1; ++x)
    {
        for(int y = -1; y <= 1; ++y)
        {
            float pcfDepth = texture(shadowMap, projCoords.xy + vec2(x, y) * texelSize).r; 
            shadow += projCoords.z - bias > pcfDepth ? 1.0 : 0.0;        
        }    
    }
    shadow /= 9.0;

    return shadow;
}

void main()
{
    vec3 color = vec3(0.8);
    vec3 normal = normalize(Normal);
    vec3 lightDir = normalize(lightPos - FragPos);

    // Standard diffuse calculation
    float diff = max(dot(normal, lightDir), 0.0);
    vec3 diffuse = diff * vec3(1.0);
    vec3 ambient = 0.15 * color;

    // Compute shadow factor (0.0 = completely lit, 1.0 = completely in shadow)
    float shadow = CalculateShadow(FragPosLightSpace, normal, lightDir);
    vec3 lighting = (ambient + (1.0 - shadow) * diffuse) * color;

    FragColor = vec4(lighting, 1.0);
}

5. Solving Common Shadow Mapping Issues

  • Shadow Acne: Caused by resolution limits where multiple fragments sample the same depth map texel. Mitigate this by applying a dynamic bias based on the angle between the light direction and the surface normal (max(0.05 * (1.0 - dot(normal, lightDir)), 0.005)).
  • Peter Panning: Applying too high of a bias causes shadows to visibly detach from objects. Counter this by keeping the bias small, or enabling front-face culling (glCullFace(GL_FRONT)) during the depth map generation pass for solid, enclosed geometry.
  • Over-sampling: Coordinates falling outside the light frustum may project invalid depth values. Setting GL_CLAMP_TO_BORDER with a border depth of 1.0 and forcing fragments with projCoords.z > 1.0 to return zero shadow resolves boundary artifacts.
