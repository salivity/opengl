Split-Sum Specular Maps in OpenGL Compute

The split-sum approximation breaks the rendering equation's specular image-based lighting integral into two independently precomputed components: the pre-filtered radiance cubemap and the 2D environment Bidirectional Reflectance Distribution Function (BRDF) integration Look-Up Table (LUT). This article details how to implement and generate both components using OpenGL compute shaders, covering importance sampling, texture bindings, mip-chain iteration, and memory synchronization for an efficient Image-Based Lighting (IBL) pipeline.

Understanding the Split-Sum Formulation

The Cook-Torrance specular radiance equation under an environment map is defined as:

\[L_o(p, \omega_o) = \int_{\Omega} f_r(p, \omega_i, \omega_o) L_i(p, \omega_i) (n \cdot \omega_i) d\omega_i\]

The split-sum approximation decouples this integral into two independent terms:

  1. Pre-filtered Environment Map: The integral of incident radiance convolved with a GGX specular lobe across multiple roughness levels: \[\int_{\Omega} L_i(p, \omega_i) d\omega_i\]
  2. Environment BRDF LUT: The integral of the specular BRDF terms (\(F, G, D\)) assuming a uniform white environment: \[\int_{\Omega} f_r(p, \omega_i, \omega_o) (n \cdot \omega_i) d\omega_i\]

Generating the Pre-Filtered Radiance Cubemap

The pre-filtered map stores blurred environment reflections across its mipmap chain. Mip level 0 stores roughness 0.0 (sharp reflections), while the maximum mip level stores roughness 1.0.

1. Cubemap Setup

Create an empty target cubemap with a full mip-chain:

GLuint prefilterMap;
glGenTextures(1, &prefilterMap);
glBindTexture(GL_TEXTURE_CUBE_MAP, prefilterMap);

int baseResolution = 512;
int maxMipLevels = 5;

for (unsigned int face = 0; face < 6; ++face) {
    glTexStorage2D(GL_TEXTURE_CUBE_MAP, maxMipLevels, GL_RGBA16F, baseResolution, baseResolution);
}
glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
glTexParameteri(GL_TEXTURE_CUBE_MAP, GL_TEXTURE_MAG_FILTER, GL_LINEAR);

2. Compute Shader Execution

Instead of rendering a screen quad for each face using framebuffers, an OpenGL compute pass handles the operation via direct 3D image writes using layered binding.

The compute shader evaluates each face and pixel:

#version 460 core
layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

layout(binding = 0) uniform samplerCube u_SourceEnvMap;
layout(binding = 0, rgba16F) uniform writeonly image2DArray u_PrefilterOut;

uniform float u_Roughness;
uniform float u_SourceResolution;

// Hammersley sequence generation
float RadicalInverse_VdC(uint bits) {
    bits = (bits << 16u) | (bits >> 16u);
    bits = ((bits & 0x55555555u) << 1u) | ((bits & 0xAAAAAAAAu) >> 1u);
    bits = ((bits & 0x33333333u) << 2u) | ((bits & 0xCCCCCCCCu) >> 2u);
    bits = ((bits & 0x0F0F0F0Fu) << 4u) | ((bits & 0xF0F0F0F0u) >> 4u);
    bits = ((bits & 0x00FF00FFu) << 8u) | ((bits & 0xFF00FF00u) >> 8u);
    return float(bits) * 2.3283064365386963e-10;
}

vec2 Hammersley(uint i, uint N) {
    return vec2(float(i) / float(N), RadicalInverse_VdC(i));
}

vec3 ImportanceSampleGGX(vec2 Xi, vec3 N, float roughness) {
    float a = roughness * roughness;
    float phi = 2.0 * 3.14159265359 * Xi.x;
    float cosTheta = sqrt((1.0 - Xi.y) / (1.0 + (a * a - 1.0) * Xi.y));
    float sinTheta = sqrt(1.0 - cosTheta * cosTheta);

    vec3 H = vec3(cos(phi) * sinTheta, sin(phi) * sinTheta, cosTheta);
    vec3 up = abs(N.z) < 0.999 ? vec3(0.0, 0.0, 1.0) : vec3(1.0, 0.0, 0.0);
    vec3 tangent = normalize(cross(up, N));
    vec3 bitangent = cross(N, tangent);

    return normalize(tangent * H.x + bitangent * H.y + N * H.z);
}

vec3 GetCubeDirection(vec2 uv, uint face) {
    uv = uv * 2.0 - 1.0;
    switch(face) {
        case 0: return normalize(vec3( 1.0, -uv.y, -uv.x));
        case 1: return normalize(vec3(-1.0, -uv.y,  uv.x));
        case 2: return normalize(vec3( uv.x,  1.0,  uv.y));
        case 3: return normalize(vec3( uv.x, -1.0, -uv.y));
        case 4: return normalize(vec3( uv.x, -uv.y,  1.0));
        case 5: return normalize(vec3(-uv.x, -uv.y, -1.0));
    }
    return vec3(0.0);
}

void main() {
    ivec3 coords = ivec3(gl_GlobalInvocationID.xyz);
    ivec2 outputSize = imageSize(u_PrefilterOut).xy;

    if (coords.x >= outputSize.x || coords.y >= outputSize.y) return;

    vec2 uv = (vec2(coords.xy) + 0.5) / vec2(outputSize);
    vec3 N = GetCubeDirection(uv, coords.z);
    vec3 R = N;
    vec3 V = R;

    const uint SAMPLE_COUNT = 1024u;
    vec3 prefilteredColor = vec3(0.0);
    float totalWeight = 0.0;

    for (uint i = 0u; i < SAMPLE_COUNT; ++i) {
        vec2 Xi = Hammersley(i, SAMPLE_COUNT);
        vec3 H = ImportanceSampleGGX(Xi, N, u_Roughness);
        vec3 L = normalize(2.0 * dot(V, H) * H - V);

        float NdotL = max(dot(N, L), 0.0);
        if (NdotL > 0.0) {
            float NdotH = max(dot(N, H), 0.0);
            float HdotV = max(dot(H, V), 0.0);
            float a = u_Roughness * u_Roughness;
            float D = (a * a) / (3.14159265359 * pow((NdotH * NdotH * (a * a - 1.0) + 1.0), 2.0));
            float pdf = (D * NdotH / (4.0 * HdotV)) + 0.0001;

            float saTexel = 4.0 * 3.14159265359 / (6.0 * u_SourceResolution * u_SourceResolution);
            float saSample = 1.0 / (float(SAMPLE_COUNT) * pdf + 0.0001);
            float mipLevel = u_Roughness == 0.0 ? 0.0 : 0.5 * log2(saSample / saTexel);

            prefilteredColor += textureLod(u_SourceEnvMap, L, mipLevel).rgb * NdotL;
            totalWeight += NdotL;
        }
    }

    prefilteredColor = prefilteredColor / totalWeight;
    imageStore(u_PrefilterOut, coords, vec4(prefilteredColor, 1.0));
}

3. Compute Dispatch Pipeline

Iterate through the mip-chain on the CPU, dispatching the compute shader once per mip level:

glUseProgram(prefilterComputeProgram);
glActiveTexture(GL_TEXTURE0);
glBindTexture(GL_TEXTURE_CUBE_MAP, sourceEnvironmentMap);

for (int mip = 0; mip < maxMipLevels; ++mip) {
    int mipWidth = baseResolution >> mip;
    int mipHeight = baseResolution >> mip;
    float roughness = (float)mip / (float)(maxMipLevels - 1);

    glUniform1f(glGetUniformLocation(prefilterComputeProgram, "u_Roughness"), roughness);
    glUniform1f(glGetUniformLocation(prefilterComputeProgram, "u_SourceResolution"), (float)baseResolution);

    // Bind layer-face views for current mip
    glBindImageTexture(0, prefilterMap, mip, GL_TRUE, 0, GL_WRITE_ONLY, GL_RGBA16F);

    GLuint groupX = (mipWidth + 15) / 16;
    GLuint groupY = (mipHeight + 15) / 16;
    glDispatchCompute(groupX, groupY, 6);

    glMemoryBarrier(GL_SHADER_IMAGE_ACCESS_BARRIER_BIT);
}

Generating the 2D Environment BRDF LUT

The BRDF LUT is a 2D look-up table parametrized by \(\cos\theta\) (view angle \(N \cdot V\)) on the x-axis and roughness on the y-axis. It stores an analytical approximation of the scale (\(R\)) and bias (\(G\)) to the Fresnel factor \(F_0\).

1. 2D Texture Allocation

A standard 512x512 16-bit float surface provides sufficient precision:

GLuint brdfLUTTexture;
glGenTextures(1, &brdfLUTTexture);
glBindTexture(GL_TEXTURE_2D, brdfLUTTexture);
glTexStorage2D(GL_TEXTURE_2D, 1, GL_RG16F, 512, 512);

glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);

2. Compute Shader Execution

The compute pass computes the geometry visibility function coupled with the Fresnel-Schlick approximation:

#version 460 core
layout(local_size_x = 16, local_size_y = 16) in;
layout(binding = 0, rg16f) uniform writeonly image2D u_BRDFLUTOut;

float GeometrySchlickGGX(float NdotV, float roughness) {
    float k = (roughness * roughness) / 2.0;
    return NdotV / (NdotV * (1.0 - k) + k);
}

float GeometrySmith(vec3 N, vec3 V, vec3 L, float roughness) {
    float NdotV = max(dot(N, V), 0.0);
    float NdotL = max(dot(N, L), 0.0);
    float ggx2 = GeometrySchlickGGX(NdotV, roughness);
    float ggx1 = GeometrySchlickGGX(NdotL, roughness);
    return ggx1 * ggx2;
}

vec2 IntegrateBRDF(float NdotV, float roughness) {
    vec3 V = vec3(sqrt(1.0 - NdotV * NdotV), 0.0, NdotV);
    vec3 N = vec3(0.0, 0.0, 1.0);

    float A = 0.0;
    float B = 0.0;
    const uint SAMPLE_COUNT = 1024u;

    for (uint i = 0u; i < SAMPLE_COUNT; ++i) {
        vec2 Xi = Hammersley(i, SAMPLE_COUNT);
        vec3 H = ImportanceSampleGGX(Xi, N, roughness);
        vec3 L = normalize(2.0 * dot(V, H) * H - V);

        float NdotL = max(L.z, 0.0);
        float NdotH = max(H.z, 0.0);
        float VdotH = max(dot(V, H), 0.0);

        if (NdotL > 0.0) {
            float G = GeometrySmith(N, V, L, roughness);
            float G_Vis = (G * VdotH) / (NdotH * NdotV);
            float Fc = pow(1.0 - VdotH, 5.0);

            A += (1.0 - Fc) * G_Vis;
            B += Fc * G_Vis;
        }
    }
    return vec2(A / float(SAMPLE_COUNT), B / float(SAMPLE_COUNT));
}

void main() {
    ivec2 coords = ivec2(gl_GlobalInvocationID.xy);
    ivec2 outputSize = imageSize(u_BRDFLUTOut);

    if (coords.x >= outputSize.x || coords.y >= outputSize.y) return;

    vec2 uv = (vec2(coords) + 0.5) / vec2(outputSize);
    vec2 integratedBRDF = IntegrateBRDF(uv.x, uv.y);

    imageStore(u_BRDFLUTOut, coords, vec4(integratedBRDF, 0.0, 0.0));
}

3. Compute Dispatch Pipeline

Dispatching the BRDF integration requires a single dispatch without mip iteration:

glUseProgram(brdfComputeProgram);
glBindImageTexture(0, brdfLUTTexture, 0, GL_FALSE, 0, GL_WRITE_ONLY, GL_RG16F);
glDispatchCompute((512 + 15) / 16, (512 + 15) / 16, 1);
glMemoryBarrier(GL_TEXTURE_FETCH_BARRIER_BIT);

Reconstruction in the PBR Shader

Once both textures are generated, sample them in the final deferred or forward evaluation pass:

vec3 PrefilteredReflection(vec3 R, float roughness) {
    float maxMipLevel = 4.0; 
    return textureLod(u_PrefilterMap, R, roughness * maxMipLevel).rgb;
}

vec3 SpecularIBL(vec3 F0, float roughness, vec3 N, vec3 V) {
    float NdotV = max(dot(N, V), 0.0);
    vec3 R = reflect(-V, N);

    vec3 prefilteredColor = PrefilteredReflection(R, roughness);
    vec2 envBRDF = texture(u_BRDFLUT, vec2(NdotV, roughness)).rg;

    return prefilteredColor * (F0 * envBRDF.x + envBRDF.y);
}

Using compute shaders for the split-sum approximation eliminates the overhead of managing specialized Framebuffer Objects (FBOs), rendering geometry passes, and switching render targets across each cubemap face and mipmap level.
